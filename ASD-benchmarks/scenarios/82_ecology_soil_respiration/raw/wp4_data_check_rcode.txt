## DATA Cleaning and Outlier Identification ##########################

# range for each selected data
sapply(data_holisoils[, c("flux", "airtemp", "t05", "t10", "t30", "tsmoisture")],
       function(x) range(x, na.rm = TRUE))
#flux airtemp  t05    t10   t30 tsmoisture
#[1,] -1.926989   -11.4 -8.0 -0.300   0.0      0.024
#[2,] 10.968533  2547.0 41.6 22.125 713.4     87.400

# Define realistic ranges 
range_temps <- c(-15, 50)
range_flux <- c(0, 5)
range_moisture <- c(0, 100)

# 1) Flag outliers
data_holisoils <- data_holisoils %>%
  mutate(outlier_airtemp = airtemp < range_temps[1] | airtemp > range_temps[2],
         outlier_t05 = t05 < range_temps[1] | t05 > range_temps[2],
         outlier_t10 = t10 < range_temps[1] | t10 > range_temps[2],
         outlier_t30 = t30 < range_temps[1] | t30 > range_temps[2],
         outlier_flux = flux < range_flux[1] | flux > range_flux[2],
         outlier_moisture = tsmoisture < range_moisture[1] | tsmoisture > range_moisture[2])

# 2) Extract rows with any outlier
outlier_rows <- data_holisoils %>%
  filter(outlier_airtemp | outlier_t05 | outlier_t10 | outlier_t30 | outlier_flux | outlier_moisture)

# Check outliers
nrow(outlier_rows)
View(outlier_rows) # Inspect outliers

#Filter to Keep Realistic Values
cleaned_data <- data_holisoils %>%
  filter((airtemp >= range_temps[1] & airtemp <= range_temps[2] | is.na(airtemp)),
         (t05 >= range_temps[1] & t05 <= range_temps[2] | is.na(t05)),
         (t10 >= range_temps[1] & t10 <= range_temps[2] | is.na(t10)),
         (t30 >= range_temps[1] & t30 <= range_temps[2] | is.na(t30)),
         (flux >= range_flux[1] & flux <= range_flux[2] | is.na(flux)),
         (tsmoisture >= range_moisture[1] & tsmoisture <= range_moisture[2] | is.na(tsmoisture)))



#Histogram for CO2 flux:
ggplot(cleaned_data, aes(x = flux)) +
  geom_histogram(binwidth = 0.1, fill = 'steelblue', color = 'black') +
  facet_wrap(~ siteid) +
  labs(title = "Histogram of CO2 fluxes by Site", x = "CO2 (g/m2/h)", y = "Count") +
  theme_bw()


#Histogram for soil moisture:
ggplot(cleaned_data, aes(x = tsmoisture)) +
  geom_histogram(binwidth = 2, fill = 'darkorange', color = 'black') +
  facet_wrap(~ siteid) +
  labs(title = "Histogram of Soil moisture by Site", x = "Soil Moisture (%)", y = "Count") +
  theme_bw()


#Reshape Temperature Data for Plotting
temp_data <- cleaned_data %>%
  select(siteid, airtemp, t05, t10, t30) %>%
  pivot_longer(cols = c(airtemp, t05, t10, t30),
               names_to = "temperature_type",
               values_to = "temperature") %>%
  filter(!is.na(temperature))

#Density plot for temperatures:
ggplot(temp_data, aes(x = temperature, color = temperature_type)) +
  geom_density() +
  facet_wrap(~ siteid) +
  labs(title = "Density plot of Temperature variables by Site",
       x = "Temperature (°C)", y = "Density", color = "Variable") +
  theme_bw()
